#include <stdio.h>

void input(int m[10][10], int r, int c)
{
    int i, j;
    printf("Enter matrix elements:\n");
    for(i = 0; i < r; i++)
        for(j = 0; j < c; j++)
            scanf("%d", &m[i][j]);
}

void display(int m[10][10], int r, int c)
{
    int i, j;
    printf("\nInput Matrix:\n");
    for(i = 0; i < r; i++)
    {
        for(j = 0; j < c; j++)
            printf("%d ", m[i][j]);
        printf("\n");
    }
}

void sparse(int m[10][10], int r, int c, int b[20][3], int *k)
{
    int i, j;

    b[0][0] = r;
    b[0][1] = c;
    *k = 1;

    for(i = 0; i < r; i++)
    {
        for(j = 0; j < c; j++)
        {
            if(m[i][j] != 0)
            {
                b[*k][0] = i;
                b[*k][1] = j;
                b[*k][2] = m[i][j];
                (*k)++;
            }
        }
    }

    b[0][2] = *k - 1;

    printf("\nSparse Matrix:\n");
    printf("Row Col Value\n");
    for(i = 0; i < *k; i++)
        printf("%d %d %d\n", b[i][0], b[i][1], b[i][2]);
}

void transpose(int b[20][3], int t[20][3], int k)
{
    int i, col, p = 1;

    t[0][0] = b[0][1];
    t[0][1] = b[0][0];
    t[0][2] = b[0][2];

    for(col = 0; col < b[0][1]; col++)
    {
        for(i = 1; i < k; i++)
        {
            if(b[i][1] == col)
            {
                t[p][0] = b[i][1];
                t[p][1] = b[i][0];
                t[p][2] = b[i][2];
                p++;
            }
        }
    }

    printf("\nSimple Transpose:\n");
    printf("Row Col Value\n");
    for(i = 0; i < p; i++)
        printf("%d %d %d\n", t[i][0], t[i][1], t[i][2]);
}

void fastTranspose(int b[20][3], int t[20][3])
{
    int rowTerms[20] = {0}, startPos[20];
    int i, j, n, numCols;

    numCols = b[0][1];
    n = b[0][2];

    t[0][0] = b[0][1];
    t[0][1] = b[0][0];
    t[0][2] = n;

    for(i = 1; i <= n; i++)
        rowTerms[b[i][1]]++;

    startPos[0] = 1;
    for(i = 1; i < numCols; i++)
        startPos[i] = startPos[i - 1] + rowTerms[i - 1];

    for(i = 1; i <= n; i++)
    {
        j = startPos[b[i][1]];
        t[j][0] = b[i][1];
        t[j][1] = b[i][0];
        t[j][2] = b[i][2];
        startPos[b[i][1]]++;
    }

    printf("\nFast Transpose:\n");
    printf("Row Col Value\n");
    for(i = 0; i <= n; i++)
        printf("%d %d %d\n", t[i][0], t[i][1], t[i][2]);
}

int main()
{
    int m[10][10], b[20][3], t[20][3];
    int r, c, k;

    printf("Enter number of rows and columns: ");
    scanf("%d%d", &r, &c);

    input(m, r, c);
    display(m, r, c);
    sparse(m, r, c, b, &k);
    transpose(b, t, k);
    fastTranspose(b, t);

    return 0;
}
