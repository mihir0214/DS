
#include <stdio.h>
#define MAX 5

int queue[MAX];
int front = -1, rear = -1;

// ENQUEUE
void enqueue() {
    int value;

    if ((rear + 1) % MAX == front) {
        printf("\nQueue Overflow!");
    }
    else {
        printf("\nEnter value: ");
        scanf("%d", &value);

        if (front == -1) {
            front = rear = 0;
        }
        else {
            rear = (rear + 1) % MAX;
        }

        queue[rear] = value;

        printf("\nInserted: %d", value);
    }
}

// DEQUEUE
void dequeue() {

    if (front == -1) {
        printf("\nQueue Underflow!");
    }
    else {
        printf("\nDeleted element: %d", queue[front]);

        if (front == rear) {
            front = rear = -1;
        }
        else {
            front = (front + 1) % MAX;
        }
    }
}

// DISPLAY
void display() {
    int i;

    if (front == -1) {
        printf("\nQueue is Empty!");
    }
    else {
        printf("\nQueue elements: ");

        i = front;

        while (1) {
            printf("%d ", queue[i]);

            if (i == rear)
                break;

            i = (i + 1) % MAX;
        }
    }
}

// MAIN FUNCTION
int main() {
    int choice;

    do {
        printf("\n\n--- CIRCULAR QUEUE ---");
        printf("\n1. Enqueue");
        printf("\n2. Dequeue");
        printf("\n3. Display");
        printf("\n4. Exit");

        printf("\nEnter your choice: ");
        scanf("%d", &choice);

        switch (choice) {

            case 1:
                enqueue();
                break;

            case 2:
                dequeue();
                break;

            case 3:
                display();
                break;

            case 4:
                printf("\nExiting...");
                break;

            default:
                printf("\nInvalid Choice!");
        }

    } while (choice != 4);

    return 0;
}
